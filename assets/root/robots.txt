# roschupka.ru — правила для поисковых роботов.
# Исходник: assets/root/robots.txt в репозитории, деплой rsync'ом в корень сайта.
#
# Открыто всё, что показываем людям, — вместе с картинками и стилями. Робот отрисовывает
# страницу так же, как браузер: закрытый CSS означает, что страница будет оценена
# по сломанному макету. Картинки блога — это его содержание, они работают на поиск
# по картинкам.
#
# Служебные каталоги движка (/blog/system/, /blog/user/) здесь не перечислены намеренно:
# сервер отдаёт на них 403, а robots.txt — не защита, а подсказка, куда не ходить.
# Дублировать им запрет незачем; стили темы при этом остаются доступны.
#
# ⚠️ Внутри группы нет ни одной пустой строки — она обрывает группу, и всё, что ниже,
# робот считает правилами без адресата, то есть не исполняет. Ровно на этом молча
# не работал прежний файл (там вдобавок пути были записаны без ведущего слэша).
#
# Что закрыто и почему:
#   sign-in, @actions — форма входа и служебные действия движка, страниц там нет;
#   main             — картинки старой главной, с 13.09.2026 корень ведёт на блог;
#   done-old         — прежняя версия портфолио;
#   openbank_test_sm — тест для коллег: ссылка рабочая, но это не содержание сайта.

User-agent: *
Disallow: /blog/sign-in/
Disallow: /blog/@actions/
Disallow: /main/
Disallow: /done-old/
Disallow: /openbank_test_sm/

# Карт две: статические страницы ведём руками, блог Эгея обновляет сама.
Sitemap: https://roschupka.ru/sitemap.xml
Sitemap: https://roschupka.ru/blog/sitemap.xml
